CREATE PROCEDURE createQuestionInstance(
    IN in_questionInstanceID INT,
    IN in_createdAt DATETIME,
    IN in_answeredAt DATETIME,
    IN in_points INT,
    IN in_questionID INT,
    IN in_gameID INT,
    IN in_username VARCHAR(255),
    IN in_userAnswer INT
)
BEGIN
    INSERT INTO QuestionInstance (
        questionInstanceID,
        createdAt,
        answeredAt,
        points,
        questionID,
        gameID,
        username,
        userAnswer
    )
    VALUES (
        in_questionInstanceID,
        in_createdAt,
        in_answeredAt,
        in_points,
        in_questionID,
        in_gameID,
        in_username,
        in_userAnswer
    );
END //

CREATE PROCEDURE AddFlashcard(
    IN in_username VARCHAR(255),
    IN in_questionID INT
)
BEGIN
    INSERT INTO Flashcard (username, questionID, priority)
    VALUES (in_username, in_questionID, 0);
END //

DELIMITER //

CREATE PROCEDURE GetUserFlashcards(
    IN in_username VARCHAR(255)
)
BEGIN
    SELECT
        f.username,
        f.questionID,
        f.priority,
        q.questionType,
        q.questionText,
        q.answerOptionID,
        q.categoryID,
        c.categoryName
    FROM Flashcard f
    JOIN Question q
        ON f.questionID = q.questionID
    JOIN Category c
        ON q.categoryID = c.categoryID
    WHERE f.username = in_username
    ORDER BY f.priority DESC, f.questionID;
END //

DELIMITER ;



DELIMITER //

CREATE PROCEDURE SearchFlashcards(
    IN p_username VARCHAR(255),
    IN in_keyword VARCHAR(255)
)
BEGIN
    SELECT
        f.username,
        f.questionID,
        f.priority,
        q.questionType,
        q.questionText,
        q.answerOptionID,
        q.categoryID,
        c.categoryName
    FROM Flashcard f
    JOIN Question q
        ON q.questionID = f.questionID
    JOIN Category c
        ON q.categoryID = c.categoryID
    WHERE f.username = p_username
        AND LOWER(q.questionText) LIKE LOWER(
            CONCAT('%', in_keyword, '%')
        )
    ORDER BY q.questionText;
END //

DELIMITER ;

CREATE PROCEDURE getGameQuestions(
    IN in_gameID INT
)
BEGIN
    SELECT
        q.questionID,
        q.questionText,
        q.answerOptionID,
        c.categoryName
    FROM GameQuestions gq
        JOIN Question q
            ON gq.questionID = q.questionID
        NATURAL JOIN Category c
        LEFT JOIN QuestionInstance qi
            ON q.questionID = qi.questionID
            AND qi.gameID = in_gameID
    WHERE gq.gameID = in_gameID
    GROUP BY q.questionID
    ORDER BY q.questionID ASC;
END //

DELIMITER ;

CREATE PROCEDURE getQuestionOptions(

CREATE TRIGGER Finished_Game_Trigger
	AFTER UPDATE ON Game
		FOR EACH ROW
	BEGIN
		DECLARE avgPts DOUBLE;
		IF old.isFinished = 0 AND new.isFinished = 1 THEN
			SELECT AVG(totalPts)	
			INTO avgPts
			FROM (
			  SELECT username, SUM(points) AS totalPts
			  FROM QuestionInstance
			  WHERE gameID = NEW.gameID
			  GROUP BY username
			) AS perUser;
    UPDATE User u
    JOIN (
      SELECT username, SUM(points) AS totalPts
      FROM QuestionInstance
      WHERE gameID = NEW.gameID
      GROUP BY username
    ) AS stats ON u.username = stats.username
    SET u.elo = u.elo + ROUND((stats.totalPts - avgPts) * 10);

  END IF;
END;

CREATE PROCEDURE initializeGame(
    IN in_categoriesCSV VARCHAR(500),
    IN in_usernamesCSV VARCHAR(1000),
    IN in_numQuestions INT,
    IN gameIDTime INT
)
BEGIN
    DECLARE n_gameID INT;

    SET TRANSACTION ISOLATION LEVEL READ COMMITTED;

    START TRANSACTION;

    INSERT INTO Game (gameDate, numQuestions, gameID)
    VALUES (NOW(), in_numQuestions, gameIDTime);

    SET n_gameID = gameIDTime;

    INSERT INTO GameSettings (categoryID, gameID)
    SELECT
        c.categoryID,
        n_gameID
    FROM Category c
    WHERE FIND_IN_SET(
        LOWER(c.categoryName),
        LOWER(in_categoriesCSV)
    );

    INSERT INTO GamePlayers (username, gameID)
    SELECT
        u.username,
        n_gameID
    FROM User u
    WHERE FIND_IN_SET(u.username, in_usernamesCSV);

    INSERT INTO GameQuestions (gameID, questionID)
    SELECT
        n_gameID,
        q.questionID
    FROM Question q
    NATURAL JOIN Category c
    WHERE FIND_IN_SET(
        LOWER(c.categoryName),
        LOWER(in_categoriesCSV)
    )
    ORDER BY RAND()
    LIMIT in_numQuestions;

    COMMIT;
END //

CREATE PROCEDURE GetUserCategoryStats(
    IN in_username VARCHAR(255)
)
BEGIN
    SELECT
        c.categoryID,
        c.categoryName,
        (correct_count / total_count) AS accuracy,
        correct_count,
        total_count
    FROM (
        SELECT
            q.categoryID,
            COUNT(*) AS correct_count
        FROM QuestionInstance qi
        NATURAL JOIN Question q
        WHERE qi.username = in_username
            AND qi.userAnswer = q.answerOptionID
        GROUP BY q.categoryID
    ) correctAnswers
    NATURAL JOIN (
        SELECT
            q.categoryID,
            COUNT(*) AS total_count
        FROM QuestionInstance qi
        NATURAL JOIN Question q
        WHERE qi.username = in_username
        GROUP BY q.categoryID
        HAVING total_count > 0
    ) totalAnswers
    JOIN Category c
        ON c.categoryID = correctAnswers.categoryID
    ORDER BY accuracy DESC;
END //
